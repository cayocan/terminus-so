# camada pessoal · ferramentas/limpar_downloads.py
@ferramenta(permissao="senha")
def limpar_downloads_antigos(dias=30):
    pasta = Path.home() / "Downloads"
    limite = agora() - timedelta(days=dias)
    antigos = [a for a in pasta.iterdir()
               if modificado(a) < limite]
    if not mostrar_lista_e_confirmar(antigos):
        return "Cancelado"
    mover_para_lixeira(antigos)
    return f"{len(antigos)} arquivos na lixeira"
